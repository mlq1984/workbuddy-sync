import{AgentBase}from"./agent-base.mjs";import{readReviewReceipt}from"../lib/review-receipt.mjs";export class ReviewAgent extends AgentBase{constructor(config={}){super({agentId:"review-agent",agentName:"Review-Agent",agentType:"specialist",capabilities:["chapter-review-receipt"],...config})}async executeTask(task){if(task.state!=="S4")throw new Error(`ReviewAgent does not support state: ${task.state}`);const{bookRoot:bookRoot,reviewReceipt:reviewReceipt}=task.payload||{};const result=readReviewReceipt(bookRoot,task.chapterId,reviewReceipt);this.publishEvent("s4.review.completed",{chapterId:task.chapterId,reviewResult:result});return result}}
