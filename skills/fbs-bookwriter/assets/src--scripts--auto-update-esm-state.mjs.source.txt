${stats.actualWordCount.toLocaleString()}`);return newState}async saveState(state){const ${this.esmStatePath}`)}getState(){if(!fs.existsSync(this.esmStatePath)){return null}try{const content=fs.readFileSync(this.esmStatePath,"utf-8");return
