#!/usr/bin/env node
import fs from"fs";import path from"path";import{fileURLToPath}from"url";import{resolveWecomDataPaths,C}from"./utils.mjs";const __dirname=path.dirname(fileURLToPath(import.meta.url));const ROOT=path.resolve(__dirname,"../../..");const{scenePacksDir:scenePacksDir}=resolveWecomDataPaths(ROOT);const LEDGER_PATH=path.join(scenePacksDir,"credits-ledger.json");const GENRE_THRESHOLDS={general:0,genealogy:100,consultant:200,ghostwriter:200,whitepaper:200,report:200,training:300,"personal-book":500};export const +${delta} 个乐包 (${CREDIT_SOURCES[source]?.label??source}) → ${ledger.balance} 个乐包${C.reset}`);return ledger.balance}export function deductCredits(source,amount,note=""){const delta=typeof 个`)}ledger.balance-=delta;ledger.total_spent+=delta;_writeLedger(ledger);_appendLog({event:"deduct",source:source,amount:delta,balance_after:ledger.balance,note:note});emitCreditsLog(`${C.yellow}[乐包] -${delta} 个乐包 (${CREDIT_SOURCES[source]?.label??source}) → ${ledger.balance} 个乐包${C.reset}`);return ledger.balance}export function checkDailyLogin(){const ledger=_readLedgerRaw();const today=(new 🎁 首次安装奖励 +${CREDIT_SOURCES.first_install.amount} 个乐包 → ${ledger.balance} 个${C.reset}`);return{added:true,balance:ledger.balance}}export function getUpgradeHint(entitlementSheet=null){const HINT_THRESHOLD=20;if(minGap>HINT_THRESHOLD){return{hint:null,nearestGenre:nearestGenre,gap:minGap,balance:balance,thresholds:thresholds}}const label=_genreLabel(nearestGenre);const hint=`🎉 再积累 ${minGap} 个乐包即可解锁「${label}」场景包！（当前 ${balance} 个乐包，门槛 ${thresholds[nearestGenre]} 个乐包）`;return{hint:hint,nearestGenre:nearestGenre,gap:minGap,balance:balance,thresholds:thresholds}}export function notifyUpgradeIfNeeded(entitlementSheet=null){const balance=getBalance();const thresholds={...GENRE_THRESHOLDS};if(Array.isArray(entitlementSheet)){for(const row of entitlementSheet){if(row.genre&&typeof row.credits_required==="number"){thresholds[row.genre]=row.credits_required}}}let nearestGenre=null;let minGap=Infinity;for(const[genre,threshold]of Object.entries(thresholds)){if(genre==="general"||threshold===0)continue;if(balance>=threshold)continue;const gap=threshold-balance;if(gap<minGap){minGap=gap;nearestGenre=genre}}if(nearestGenre===null){return{level:"upsell",message:`🎊 所有场景包体裁均已解锁！如需激活高级定制能力，可使用激活码升级到企业增强版。`}}const ${thresholds[nearestGenre]} 个）`}}if(minGap<=50){return{level:"mid",message:`📈 再完成约 ${chapters} 章即可解锁「${label}」场景包（还差 ${minGap} 个乐包，当前 ${balance} 个）`}}return{level:"none",message:null}}export function formatBalanceSummary(entitlementSheet=null){const ledger=readLedger();const balance=ledger.balance;const thresholds={...GENRE_THRESHOLDS};if(Array.isArray(entitlementSheet)){for(const row of entitlementSheet){if(row.genre&&typeof row.credits_required==="number"){thresholds[row.genre]=row.credits_required}}}const lines=[`**当前乐包余额：${balance} 个**`,""];lines.push("| 场景包体裁 | 门槛 | 状态 |");lines.push("|----------|------|------|");for(const[genre,threshold]of Object.entries(thresholds)){if(genre==="general")continue;const label=_genreLabel(genre);const status=balance>=threshold?"✅
