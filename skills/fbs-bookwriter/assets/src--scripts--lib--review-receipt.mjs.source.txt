import fs from"node:fs";import path from"node:path";import crypto from"node:crypto";export const REVIEW_SCHEMA="fbs-bookwriter.chapter-review/v1";export function readBookFile(bookRoot,filePath){const root=fs.realpathSync(bookRoot);const target=fs.realpathSync(path.resolve(bookRoot,filePath));const relative=path.relative(root,target);if(!relative||relative.startsWith(".."+path.sep)||relative===".."||path.isAbsolute(relative)){throw new Error("审校文件必须位于本书目录内")}if(!fs.statSync(target).isFile())throw new Error("审校目标不是文件");const bytes=fs.readFileSync(target);return{path:target,bytes:bytes,sha256:crypto.createHash("sha256").update(bytes).digest("hex")}}export function readReviewReceipt(bookRoot,chapterId,receiptPath){const file=receiptPath||path.join(".fbs","reviews",chapterId+".json");let receipt;try{receipt=JSON.parse(readBookFile(bookRoot,file).bytes.toString("utf8").replace(/^\uFEFF/,""))}catch(error){throw new Error(`无法读取章节审校回执；请先完成审校并提供 ${file}。${error.message}`)}if(receipt.schema!==REVIEW_SCHEMA||receipt.chapterId!==chapterId||receipt.decision!=="approved"){throw new Error("审校回执类型、章节或批准状态不匹配")}if(!["content","compliance","business"].every(key=>receipt.checks?.[key]==="passed")){throw new Error("审校回执尚未完成内容、合规与业务检查")}if(typeof receipt.source!=="string"||path.isAbsolute(receipt.source)||!/^[a-f0-9]{64}$/.test(receipt.sourceSha256||"")){throw new Error("审校回执缺少本书相对来源路径或 SHA256")}const source=readBookFile(bookRoot,receipt.source);if(!source.bytes.length||source.sha256!==receipt.sourceSha256){throw new Error("章节内容已变化或为空，需要重新审校")}return{chapterId:chapterId,bookRoot:bookRoot,passed:true,status:"recorded",sourcePath:source.path,sourceSha256:source.sha256,checks:receipt.checks,evidenceBasis:"caller-supplied-review-receipt",reviewerIdentityVerified:false,issues:[],suggestions:[]}}export function readReviewedSource(bookRoot,review){if(!review?.passed||typeof review.sourcePath!=="string"||!/^[a-f0-9]{64}$/.test(review.sourceSha256||"")){throw new Error("交付缺少与章节字节绑定的审校结果")}const source=readBookFile(bookRoot,review.sourcePath);if(source.sha256!==review.sourceSha256)throw new Error("审校后章节已变化，停止交付");return source}
