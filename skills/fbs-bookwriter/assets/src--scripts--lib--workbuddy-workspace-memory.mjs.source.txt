import fs from"fs";import path from"path";function todayTag(){return(new Date).toISOString().slice(0,10)}export function appendWorkbuddyMemoryMirror({bookRoot:bookRoot,title:title="FBS session 镜像",lines:lines,quiet:quiet=false}){const root=path.resolve(bookRoot);const memDir=path.join(root,".workbuddy","memory");const file=path.join(memDir,`${todayTag()}.md`);try{fs.mkdirSync(memDir,{recursive:true});const block=["",`### ${title} — ${(new Date).toISOString()}`,"",...lines.map(l=>l.startsWith("-")||l.startsWith("|")?l:`- ${l}`),""].join("\n");fs.appendFileSync(file,block,"utf8");if(!quiet)console.log(`[workbuddy-memory] 已追加叙事层：${file}`);return{path:file,appended:true}}catch(e){if(!quiet)console.warn("[workbuddy-memory] 追加失败（不阻断退出）：",e.message);return{path:null,appended:false}}}
