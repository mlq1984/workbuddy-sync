#!/usr/bin/env node
import fs from"fs";import path from"path";import{pathToFileURL}from"url";const VALID=new Set(["IDLE","INTAKE","RESEARCH","PLAN","WRITE","REVIEW","WRITE_MORE","DELIVER"]);const MIN_NORM_SKELETON=`# 规范执行状态（.fbs/规范执行状态.md）\n\n> 由 \`init-fbs-multiagent-artifacts.mjs\` 生成完整模板；本骨架仅供 CLI 首次写入前兜底。\n\n## ESM 状态追踪\n\n| 时间 | 旧状态 | 新状态 | 触发原因 | 出口条件 |\n|------|--------|--------|---------|---------|\n| （待补） | — | — | — | — |\n\n## 切换日志\n\n`;function parseArgs(argv){const options={bookRoot:null,from:null,to:null,reason:"",genre:""};for(let i=2;i<argv.length;i++){const arg=argv[i];if(arg==="--book-root")options.bookRoot=argv[++i];else if(arg==="--from")options.from=argv[++i];else if(arg==="--to")options.to=argv[++i];else ensureNormExecState(fbsDir,{logger:logger=console,quiet:quiet=false}={}){const targetPath=path.join(fbsDir,"规范执行状态.md");if(!fs.existsSync(targetPath)){fs.mkdirSync(fbsDir,{recursive:true});fs.writeFileSync(targetPath,MIN_NORM_SKELETON,"utf8");if(!quiet)logger.log("create skeleton:",targetPath)}return targetPath}function prependTransitionLog(normPath,line){let body=fs.readFileSync(normPath,"utf8");const marker="## 切换日志";const index=body.indexOf(marker);if(index===-1){body+=`\n\n${marker}\n\n${line}\n`;fs.writeFileSync(normPath,body,"utf8");return}const afterHeader=body.indexOf("\n",index);const insertAt=afterHeader===-1?index+marker.length:afterHeader+1;body=body.slice(0,insertAt)+line+"\n"+body.slice(insertAt);fs.writeFileSync(normPath,body,"utf8")}const ITER_DEFAULTS={iterationPhase:"none",expansionRound:0,refinementRound:0,lastInterruptAt:"",lastInterruptReason:""};function readIterationFromExisting(fbsDir){const p=path.join(fbsDir,"esm-state.md");if(!fs.existsSync(p))return{...ITER_DEFAULTS};const t=fs.readFileSync(p,"utf8");const pickStr=key=>{const m=new RegExp(`^${key}:\\s*(.+)$`,"m").exec(t);if(!m)return null;let v=m[1].trim();if(v.startsWith('"')&&v.endsWith('"')){try{return JSON.parse(v)}catch{return v.slice(1,-1)}}return v};const pickNum=(key,def)=>{const m=new RegExp(`^${key}:\\s*(\\d+)`,"m").exec(t);return m?Number(m[1]):def};const ip=pickStr("iterationPhase");return{iterationPhase:ip!=null&&ip!==""?String(ip):ITER_DEFAULTS.iterationPhase,expansionRound:pickNum("expansionRound",ITER_DEFAULTS.expansionRound),refinementRound:pickNum("refinementRound",ITER_DEFAULTS.refinementRound),lastInterruptAt:String(pickStr("lastInterruptAt")??""),lastInterruptReason:String(pickStr("lastInterruptReason")??"")}}function writeEsmState(fbsDir,{from:from,to:to,reason:reason,genre:genre,iso:iso,iteration:iteration},{logger:logger=console,quiet:quiet=false}={}){const it=iteration||{...ITER_DEFAULTS};const body=`---\ncurrentState: "${to}"\npreviousState: "${from}"\nlastTransitionAt: "${iso}"\ntransitionReason: ${JSON.stringify(reason||"")}\ngenre: ${JSON.stringify(genre||"")}\niterationPhase: ${JSON.stringify(String(it.iterationPhase||"none"))}\nexpansionRound: ${Number(it.expansionRound)||0}\nrefinementRound: ${Number(it.refinementRound)||0}\nlastInterruptAt: ${JSON.stringify(it.lastInterruptAt||"")}\nlastInterruptReason: ${JSON.stringify(it.lastInterruptReason||"")}\nmaintainedBy: "scripts/fbs-record-esm-transition.mjs"\n---\n\n# ESM 当前状态（.fbs/esm-state.md）\n\n> **权威流程**：见 \`references/01-core/section-3-workflow.md\`「执行状态机」。\n> **更新**：每次对话内输出状态切换宣告后，**应**运行本脚本或等价更新本文件，使磁盘与对话一致（v1.8.0 审计：外部可验证）。\n> **迭代子阶段**（扩写/精修）：见 \`references/01-core/s3-expansion-phase.md\`、\`references/01-core/s3-refinement-phase.md\`；\`iterationPhase\` 为 \`none\` / \`expansion\` / \`refinement\`。\n\n| 字段 | 值 |\n|------|-----|\n| 当前状态 | **${to}** |\n| 上一状态 | ${from} |\n| 切换时间 | ${iso} |\n| 原因 | ${reason||"—"} |\n| 体裁等级 | ${genre||"—"} |\n| **iterationPhase** | ${it.iterationPhase||"none"} |\n| **expansionRound** | ${it.expansionRound??0} |\n| **refinementRound** | ${it.refinementRound??0} |\n| **lastInterruptAt** | ${it.lastInterruptAt||"—"} |\n`;const targetPath=path.join(fbsDir,"esm-state.md");fs.writeFileSync(targetPath,body,"utf8");if(!quiet)logger.log("write:",targetPath);return targetPath}const STAGE_ALIAS_TO_LEGACY={S0:"INTAKE",S1:"RESEARCH",S2:"PLAN",S3:"WRITE",S4:"REVIEW",S5:"DELIVER",S6:"DELIVER"};export function normalizeStateName(value){return String(value||"").trim().toUpperCase().replace(/[\s-]+/g,"_")}export function resolveStateAlias(value){const raw=normalizeStateName(value);if(VALID.has(raw))return raw;if(STAGE_ALIAS_TO_LEGACY[raw])return STAGE_ALIAS_TO_LEGACY[raw];return null}function validateTransitionInput({from:from,to:to}){const fromState=resolveStateAlias(from);const recordESMTransition({bookRoot:bookRoot=process.cwd(),from:from,to:to,reason:reason="",genre:genre="",timestamp:timestamp,logger:logger=console,quiet:quiet=false}={}){if(!from||!to){throw new Error("缺少 from/to，无法记录 ESM 状态切换")}const{fromState:fromState,toState:toState}=validateTransitionInput({from:from,to:to});const root=path.resolve(bookRoot||process.cwd());const fbsDir=path.join(root,".fbs");fs.mkdirSync(fbsDir,{recursive:true});const iso=timestamp||(new Date).toISOString();const normalizedGenre=String(genre||"").trim().toUpperCase();const iteration=readIterationFromExisting(fbsDir);const esmStatePath=writeEsmState(fbsDir,{from:fromState,to:toState,reason:reason,genre:normalizedGenre,iso:iso,iteration:iteration},{logger:logger,quiet:quiet});const normPath=ensureNormExecState(fbsDir,{logger:logger,quiet:quiet});const logLine=`- **${iso}** · \`${fromState}\` → \`${toState}\` · ${reason||"—"}${normalizedGenre?` · 体裁 log:",normPath);logger.log("done.")}return{root:root,fbsDir:fbsDir,esmStatePath:esmStatePath,normPath:normPath,from:fromState,to:toState,timestamp:iso,genre:normalizedGenre,logLine:logLine}}export async function main(rawArgs=parseArgs(process.argv)){const{bookRoot:bookRoot,from:from,to:to,reason:reason,genre:genre}=rawArgs;if(!bookRoot||!from||!to){console.error('用法: node
