_emptyConfig(){return{_version:"2.0",_comment:"用户个人配置（自动生成，勿手动编辑 books 数组）。此文件不含密钥，但含本机路径信息，已加入 .gitignore。",wecom_userid:"",wecom_corp_id:"",books:[]}}function normalizeUserConfig(config){const cap=Number.isFinite(MAX_USER_BOOKS)&&MAX_USER_BOOKS>0?MAX_USER_BOOKS:300;base.books=deduped.slice(0,cap);return base}
